# Python extension support for folly.
# This builds the folly Python bindings (iobuf, executor, etc.) for use by
# downstream projects like thrift-python.

# Create tree of symbolic links in structure required for successful
# compilation by Cython.
#   - must be in path named same as extension

set(_cybld "${CMAKE_CURRENT_BINARY_DIR}/cybld")

# ---------------------------------------------------------------------------
# Build the Cython staging tree
# ---------------------------------------------------------------------------

file(
  GLOB_RECURSE RelativeBindingFiles
  CONFIGURE_DEPENDS
  LIST_DIRECTORIES false
  RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  "*.pyx"
  "*.pxd"
  "*.pyi"
  "*.py"
  "*.h"
  "*.cpp"
)

# Tests and setup.py do not belong inside the staged Python package.
list(FILTER RelativeBindingFiles EXCLUDE REGEX "^test/")
list(FILTER RelativeBindingFiles EXCLUDE REGEX "^setup\\.py$")

file(
  GLOB_RECURSE CythonSources
  CONFIGURE_DEPENDS
  LIST_DIRECTORIES false
  RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  "*.pyx"
)

list(FILTER CythonSources EXCLUDE REGEX "^test/")

# A foo.pyx module generates foo.cpp. Do not stage a handwritten foo.cpp
# at the same destination.
foreach(_pyx IN LISTS CythonSources)
  string(REGEX REPLACE "\\.pyx$" ".cpp" _generated_cpp "${_pyx}")
  list(REMOVE_ITEM RelativeBindingFiles "${_generated_cpp}")
endforeach()

file(MAKE_DIRECTORY "${_cybld}/folly")
file(MAKE_DIRECTORY "${_cybld}/folly/python")
file(MAKE_DIRECTORY "${_cybld}/folly/windows")
file(MAKE_DIRECTORY "${_cybld}/folly/python/windows")

# Remove files left by the previous staging layout. In particular,
# cybld/folly/executor.h can shadow folly/Executor.h on a
# case-insensitive macOS filesystem.
file(
  GLOB _legacy_root_headers
  LIST_DIRECTORIES false
  "${_cybld}/folly/*.h"
)

foreach(_legacy_header IN LISTS _legacy_root_headers)
  get_filename_component(_legacy_name "${_legacy_header}" NAME)

  if(NOT _legacy_name STREQUAL "iobuf_api.h")
    file(REMOVE "${_legacy_header}")
  endif()
endforeach()

# Remove stale handwritten symlinks at paths reserved for Cython output.
foreach(_pyx IN LISTS CythonSources)
  string(REGEX REPLACE "\\.pyx$" ".cpp" _generated_cpp "${_pyx}")
  file(REMOVE "${_cybld}/folly/${_generated_cpp}")
endforeach()

# Package markers.
file(
  WRITE "${_cybld}/folly/__init__.py"
  "# Auto-generated by CMake\n"
)

file(
  WRITE "${_cybld}/folly/python/__init__.py"
  "# Auto-generated by CMake\n"
)

file(
  WRITE "${_cybld}/folly/windows/__init__.py"
  "# Auto-generated by CMake\n"
)

file(
  WRITE "${_cybld}/folly/py.typed"
  "# Marker file for PEP 561\n"
)

file(
  WRITE "${_cybld}/folly/python/py.typed"
  "# Marker file for PEP 561\n"
)

set(BindingSourcePaths)

foreach(_rel IN LISTS RelativeBindingFiles)
  set(_src "${CMAKE_CURRENT_SOURCE_DIR}/${_rel}")
  list(APPEND BindingSourcePaths "${_src}")

  # Cython package inputs live under cybld/folly.
  #
  # Binding C++ headers live under cybld/folly/python because they are
  # included as <folly/python/...>. Keeping executor.h out of cybld/folly
  # prevents it from shadowing the core <folly/Executor.h> header.
  if(_rel MATCHES "\\.h$")
    set(_dst "${_cybld}/folly/python/${_rel}")
  else()
    set(_dst "${_cybld}/folly/${_rel}")
  endif()

  get_filename_component(_dst_dir "${_dst}" DIRECTORY)
  file(MAKE_DIRECTORY "${_dst_dir}")

  message(STATUS "Staging Python binding file: ${_rel}")
  message(STATUS "Linking ${_src} to ${_dst}")

  file(
    CREATE_LINK
    "${_src}"
    "${_dst}"
    SYMBOLIC
    RESULT _link_result
  )

  if(NOT _link_result STREQUAL "0")
    message(
      FATAL_ERROR
      "Could not create binding symlink:\n"
      "  source: ${_src}\n"
      "  target: ${_dst}\n"
      "  error:  ${_link_result}"
    )
  endif()
endforeach()

# ---------------------------------------------------------------------------
# Build flags used by setup.py
# ---------------------------------------------------------------------------

set(prop "$<TARGET_PROPERTY:folly,INCLUDE_DIRECTORIES>")
set(incs "$<$<BOOL:${prop}>:-I$<JOIN:${prop},:>>")
set(glog_lib "${GLOG_LIBRARY}")
cmake_path(REMOVE_FILENAME glog_lib)
set(libs "-L${CMAKE_CURRENT_BINARY_DIR}:${CMAKE_BINARY_DIR}:${glog_lib}")

if(APPLE)
  string(
    CONCAT DEFAULT_SETUP_LDFLAGS
    "-Wl,-rpath,${CMAKE_INSTALL_PREFIX}/${LIB_INSTALL_DIR} "
    "-Wl,-undefined,dynamic_lookup "
    "-Wl,-rpath,${CMAKE_CURRENT_BINARY_DIR} "
    "-Wl,-rpath,${CMAKE_BINARY_DIR}"
  )
else()
  string(
    CONCAT DEFAULT_SETUP_LDFLAGS
    "-Wl,--no-as-needed "
    "-Wl,-rpath,${CMAKE_CURRENT_BINARY_DIR}:${CMAKE_BINARY_DIR}"
  )
endif()

if(DEFINED ENV{LDFLAGS} AND NOT "$ENV{LDFLAGS}" STREQUAL "")
  set(SETUP_LDFLAGS "$ENV{LDFLAGS}")

  message(
    STATUS
    "Using user-provided LDFLAGS for folly/python: ${SETUP_LDFLAGS}"
  )
else()
  set(SETUP_LDFLAGS "${DEFAULT_SETUP_LDFLAGS}")

  message(
    STATUS
    "Using default LDFLAGS for folly/python: ${SETUP_LDFLAGS}"
  )
endif()

# ---------------------------------------------------------------------------
# Generate Cython API headers
# ---------------------------------------------------------------------------

add_custom_command(
  OUTPUT "${_cybld}/folly/iobuf_api.h"

  # Prevent a failed Cython invocation from appearing successful because
  # an old output file remains.
  COMMAND
    "${CMAKE_COMMAND}" -E rm -f
    "${_cybld}/folly/iobuf_api.h"

  COMMAND
    ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/setup.py" --api-only

  # setup.py currently does not reliably propagate every Cython error.
  # Verify that the declared output was actually generated.
  COMMAND
    ${Python3_EXECUTABLE} -c "from pathlib import Path; p = Path('${_cybld}/folly/iobuf_api.h'); assert p.is_file(), f'Cython did not generate {p}'"

  DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/setup.py"
    ${BindingSourcePaths}

  WORKING_DIRECTORY "${_cybld}"
  COMMENT "Generating folly/iobuf_api.h"
  VERBATIM
)

add_custom_target(
  folly_python_api_headers ALL
  DEPENDS "${_cybld}/folly/iobuf_api.h"
)

# Handwritten C++ code includes <folly/python/iobuf_api.h>, while Cython
# generates folly/iobuf_api.h.
add_custom_command(
  OUTPUT "${_cybld}/folly/python/iobuf_api.h"

  COMMAND
    "${CMAKE_COMMAND}" -E make_directory
    "${_cybld}/folly/python"

  COMMAND
    "${CMAKE_COMMAND}" -E rm -f
    "${_cybld}/folly/python/iobuf_api.h"

  COMMAND
    "${CMAKE_COMMAND}" -E create_symlink
    "${_cybld}/folly/iobuf_api.h"
    "${_cybld}/folly/python/iobuf_api.h"

  DEPENDS
    "${_cybld}/folly/iobuf_api.h"

  COMMENT "Linking folly/python/iobuf_api.h"
  VERBATIM
)

add_custom_target(
  create_post_binding_symlink ALL
  DEPENDS "${_cybld}/folly/python/iobuf_api.h"
)

# ---------------------------------------------------------------------------
# Common C++ support library
# ---------------------------------------------------------------------------

# ProactorExecutor.cpp, fibers.cpp, and iobuf_ext.cpp are compiled by their
# corresponding setuptools Extension definitions. Do not compile them again
# into folly_python_cpp.
add_library(
  folly_python_cpp SHARED
    AsyncioExecutor.cpp
    GILAwareManualExecutor.cpp
    ProactorExecutor.cpp
    error.cpp
    executor.cpp
    fibers.cpp
    iobuf.cpp
    iobuf_ext.cpp
)

target_include_directories(
  folly_python_cpp
  PRIVATE
    "${Python3_INCLUDE_DIRS}"
    "${_cybld}"
)

target_link_libraries(
  folly_python_cpp
  PRIVATE
    Python3::Python
  PUBLIC
    folly
)

add_dependencies(
  folly_python_cpp
  folly
  create_post_binding_symlink
)

set_property(
  TARGET folly_python_cpp
  PROPERTY VERSION "${PACKAGE_VERSION}"
)

target_compile_definitions(
  folly_python_cpp
  PRIVATE
    BOOST_NO_AUTO_PTR
)

apply_folly_compile_options_to_target(folly_python_cpp)

# ---------------------------------------------------------------------------
# Build the extension modules and wheel
# ---------------------------------------------------------------------------

add_custom_target(
  folly_python_bindings ALL
  DEPENDS
    folly_python_cpp
    folly_python_api_headers
)

add_custom_command(
  TARGET folly_python_bindings
  POST_BUILD

  # Build the extension modules with the include and library paths obtained
  # from the CMake targets.
  COMMAND
    "${CMAKE_COMMAND}" -E env
    "CC=${CMAKE_C_COMPILER}"
    "CXX=${CMAKE_CXX_COMPILER}"
    "LDFLAGS=${SETUP_LDFLAGS}"
    ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/setup.py" build_ext
    -f ${incs} ${libs}

  # Ensure installation cannot accidentally select an old wheel.
  COMMAND
    "${CMAKE_COMMAND}" -E rm -rf
    "${_cybld}/dist"

  COMMAND
    "${CMAKE_COMMAND}" -E env
    "CC=${CMAKE_C_COMPILER}"
    "CXX=${CMAKE_CXX_COMPILER}"
    "LDFLAGS=${SETUP_LDFLAGS}"
    ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/setup.py" bdist_wheel

  WORKING_DIRECTORY "${_cybld}"
  COMMENT "Building Folly Python extensions and wheel"
  VERBATIM
)

# ---------------------------------------------------------------------------
# Install the C++ support library and development files
# ---------------------------------------------------------------------------

install(
  TARGETS folly_python_cpp
  EXPORT folly
  DESTINATION "${LIB_INSTALL_DIR}"
)

install(

  DESTINATION "${INCLUDE_INSTALL_DIR}/folly/python"
  FILES_MATCHING
    PATTERN "*.pxd"

    PATTERN "*.pyi"

install(

  DESTINATION "${INCLUDE_INSTALL_DIR}/folly/python"
  COMPONENT dev
)

# ---------------------------------------------------------------------------
# Install the wheel
# ---------------------------------------------------------------------------

set(INSTALL_DIR "${CMAKE_INSTALL_PREFIX}")

if(
  DEFINED PYTHON_PACKAGE_INSTALL_DIR
  AND NOT "${PYTHON_PACKAGE_INSTALL_DIR}" STREQUAL ""
)
  set(INSTALL_DIR "${PYTHON_PACKAGE_INSTALL_DIR}")
endif()

install(CODE "
  set(_cybld ${_cybld})
  set(INSTALL_DIR ${INSTALL_DIR})
  set(Python3_EXECUTABLE ${Python3_EXECUTABLE})
")

install(CODE [=[
  file(
    GLOB _folly_wheels
    "${_cybld}/dist/folly-*.whl"
  )

  list(LENGTH _folly_wheels _folly_wheel_count)

  if(NOT _folly_wheel_count EQUAL 1)
    message(
      FATAL_ERROR
      "Expected exactly one Folly wheel in ${_cybld}/dist, found ${_folly_wheel_count}: ${_folly_wheels}"
    )
  endif()

  list(GET _folly_wheels 0 _folly_wheel)

  # Custom install scripts must apply DESTDIR explicitly.
  set(_pip_prefix "$ENV{DESTDIR}${INSTALL_DIR}")

  execute_process(
    COMMAND
      ${Python3_EXECUTABLE} -m pip install ${_folly_wheel}
      --prefix ${_pip_prefix}
      --no-deps
      --force-reinstall
    COMMAND_ECHO STDOUT
    RESULT_VARIABLE _pip_result
  )

  if(NOT _pip_result STREQUAL "0")
    message(
      FATAL_ERROR "pip failed to install ${_folly_wheel}: exit code ${_pip_result}"
    )
  endif()
]=])
